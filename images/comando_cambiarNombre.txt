Que todas las imagenes de la carpeta se llamen box1.jpg, box2.jpg ...

i=1
for f in *.jpg; do
    mv "$f" "box$i.${f##*.}"
    i=$((i+1))
done


eliminar la palabra box del nombre para que se llamen 1.jpg, 2.jpg ...

for f in box*; do mv "$f" "${f#box}"; done

Renombrar todas las imágenes de una carpeta a números consecutivos

i=1 for f in *.jpg; 
do mv "$f" "$i.jpg" i=$((i+1))
